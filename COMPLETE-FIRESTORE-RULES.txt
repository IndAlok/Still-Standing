rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Users collection - allow reading all users, writing own profile
    match /users/{userId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.firebaseUID;
      allow update: if request.auth != null && 
        (request.auth.uid == userId || request.auth.uid == resource.data.firebaseUID);
      allow delete: if request.auth != null && 
        (request.auth.uid == userId || request.auth.uid == resource.data.firebaseUID);
    }
    
    // Crews collection - comprehensive access control
    match /crews/{crewId} {
      // Allow creation if user is the creator being set
      allow create: if request.auth != null && 
        request.auth.uid == request.resource.data.createdBy &&
        request.auth.uid in request.resource.data.members;
      
      // Allow reading public crews or crews user is member of
      allow read: if request.auth != null && 
        (resource.data.isPublic == true || 
         request.auth.uid in resource.data.members ||
         request.auth.uid == resource.data.createdBy);
      
      // Allow updates if user is creator or member (for joining/leaving)
      allow update: if request.auth != null && 
        (request.auth.uid == resource.data.createdBy || 
         request.auth.uid in resource.data.members ||
         request.auth.uid in request.resource.data.members);
      
      // Allow deletion only by creator
      allow delete: if request.auth != null && 
        request.auth.uid == resource.data.createdBy;
    }
    
    // Memberships collection - users manage their own memberships
    match /memberships/{membershipId} {
      // Allow creation if user is creating their own membership
      allow create: if request.auth != null && 
        request.auth.uid == request.resource.data.userId;
      
      // Allow reading own memberships or if user is crew admin
      allow read: if request.auth != null && 
        (request.auth.uid == resource.data.userId ||
         exists(/databases/$(database)/documents/crews/$(resource.data.crewId)) &&
         get(/databases/$(database)/documents/crews/$(resource.data.crewId)).data.createdBy == request.auth.uid);
      
      // Allow updating/deleting own memberships or by crew admin
      allow update, delete: if request.auth != null && 
        (request.auth.uid == resource.data.userId ||
         exists(/databases/$(database)/documents/crews/$(resource.data.crewId)) &&
         get(/databases/$(database)/documents/crews/$(resource.data.crewId)).data.createdBy == request.auth.uid);
    }
    
    // Messages collection - crew members can create and read messages
    match /messages/{messageId} {
      // Allow creation if user is the sender and member of the crew
      allow create: if request.auth != null && 
        request.auth.uid == request.resource.data.senderId &&
        exists(/databases/$(database)/documents/crews/$(request.resource.data.crewId)) &&
        (request.auth.uid in get(/databases/$(database)/documents/crews/$(request.resource.data.crewId)).data.members ||
         get(/databases/$(database)/documents/crews/$(request.resource.data.crewId)).data.createdBy == request.auth.uid);
      
      // Allow reading messages if user is member of the crew
      allow read: if request.auth != null &&
        exists(/databases/$(database)/documents/crews/$(resource.data.crewId)) &&
        (request.auth.uid in get(/databases/$(database)/documents/crews/$(resource.data.crewId)).data.members ||
         get(/databases/$(database)/documents/crews/$(resource.data.crewId)).data.createdBy == request.auth.uid);
      
      // Allow updating/deleting own messages
      allow update, delete: if request.auth != null && 
        request.auth.uid == resource.data.senderId;
    }
  }
}
